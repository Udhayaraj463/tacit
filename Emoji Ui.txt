  const scanLiveLocationNow = async (isSilentBackground = false) => {
    if (!isOnline || isAutoDetectingLocation || isGeneratingVenue) return;
    setIsAutoDetectingLocation(true);
    if (!isSilentBackground) {
      setMochiMood('thinking');
      setMochiSpeech('Scanning satellites & nearby buildings... 🛰️');
    }

    try {
      let { status } = await Location.requestForegroundPermissionsAsync();
      if (status !== 'granted') {
        if (!isSilentBackground) setMochiSpeech('Location permission needed! Enable GPS in browser settings. 📍');
        setIsAutoDetectingLocation(false);
        return;
      }

      const location = await Location.getCurrentPositionAsync({ accuracy: Location.Accuracy.Balanced });
      const { latitude, longitude } = location.coords;

      let detectedName = '';
      let detectedCategory = '';

      // Tier 1: Free, CORS-safe OpenStreetMap Overpass API (Finds exact POIs like Library, 7-Eleven, Cafe within 80m)
      try {
        const overpassQuery = `[out:json][timeout:6];(node(around:80,${latitude},${longitude})["name"];way(around:80,${latitude},${longitude})["name"];);out body 5;`;
        const osmRes = await fetch(`https://overpass-api.de/api/interpreter?data=${encodeURIComponent(overpassQuery)}`);
        const osmData = await osmRes.json();
        if (osmData?.elements?.length > 0) {
          const poi = osmData.elements.find((el: any) => el.tags && (el.tags.amenity || el.tags.shop || el.tags.building || el.tags.tourism)) || osmData.elements[0];
          if (poi?.tags) {
            detectedName = poi.tags['name:en'] || poi.tags.name || '';
            detectedCategory = poi.tags.amenity || poi.tags.shop || poi.tags.building || '';
          }
        }
      } catch (e) {}

      // Tier 2: OpenStreetMap Nominatim Reverse Geocode Fallback
      if (!detectedName) {
        try {
          const nomRes = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${latitude}&lon=${longitude}&zoom=18&addressdetails=1`);
          const nomData = await nomRes.json();
          detectedName = nomData?.name || nomData?.address?.amenity || nomData?.address?.shop || nomData?.address?.building || nomData?.address?.Get || '';
          detectedCategory = nomData?.type || nomData?.category || '';
        } catch (e) {}
      }

      if (!detectedName) {
        if (!isSilentBackground) setMochiSpeech(`GPS locked (${latitude.toFixed(3)}, ${longitude.toFixed(3)}), no specific counter detected—tap '+' to type it!`);
        setIsAutoDetectingLocation(false);
        return;
      }

      if (isSilentBackground && lastLocationIdRef.current === detectedName) {
        setIsAutoDetectingLocation(false);
        return;
      }
      lastLocationIdRef.current = detectedName;

      const combinedCheck = `${detectedName} ${detectedCategory}`.toLowerCase();
      const currentVenues = venuesRef.current;

      // Smart Match against existing dock venues first
      let matchedVenue = currentVenues.find(v => v.nameKey.toLowerCase() === detectedName.toLowerCase());
      if (!matchedVenue) {
        if (combinedCheck.includes('library') || combinedCheck.includes('圖書館')) matchedVenue = currentVenues.find(v => v.id === 'library');
        else if (combinedCheck.includes('7-eleven') || combinedCheck.includes('familymart') || combinedCheck.includes('mart') || combinedCheck.includes('convenience') || combinedCheck.includes('超商')) matchedVenue = currentVenues.find(v => v.id === 'convenience');
        else if (combinedCheck.includes('hospital') || combinedCheck.includes('clinic') || combinedCheck.includes('醫') || combinedCheck.includes('診所')) matchedVenue = currentVenues.find(v => v.id === 'hospital');
        else if (combinedCheck.includes('cafe') || combinedCheck.includes('coffee') || combinedCheck.includes('tea') || combinedCheck.includes('咖啡')) matchedVenue = currentVenues.find(v => v.id === 'cafe');
        else if (combinedCheck.includes('station') || combinedCheck.includes('metro') || combinedCheck.includes('train') || combinedCheck.includes('捷運') || combinedCheck.includes('車站')) matchedVenue = currentVenues.find(v => v.id === 'transit');
        else if (combinedCheck.includes('restaurant') || combinedCheck.includes('food') || combinedCheck.includes('餐廳')) matchedVenue = currentVenues.find(v => v.id === 'restaurant');
      }

      if (matchedVenue) {
        setActiveVenue(matchedVenue);
        setMochiMood('excited');
        setMochiSpeech(`📍 Radar locked onto "${detectedName}"! Switched to ${matchedVenue.short} mode.`);
        setLiveHeardTranslated(`📍 Live GPS Detected: ${detectedName}`);
      } else {
        // Brand new place! Generate custom AI cards automatically
        setMochiMood('thinking');
        setMochiSpeech(`✨ Discovered "${detectedName}"! Crafting custom cards for you...`);
        const generated = await fetchDynamicVenueCards(detectedName);
        if (generated && generated.options?.length > 0) {
          const updatedVenues = [generated, ...currentVenues];
          setVenues(updatedVenues);
          setActiveVenue(generated);
          setMochiMood('excited');
          setMochiSpeech(generated.vibeQuote || `🎉 Welcome to ${detectedName}! Fresh cards are ready.`);
          setLiveHeardTranslated(`✨ AI Auto-Created Spot: ${detectedName}`);
          await AsyncStorage.setItem('@aac_custom_venues_v2', JSON.stringify(updatedVenues));
        }
      }
    } catch (err) {
      if (!isSilentBackground) setMochiSpeech('Could not reach GPS satellites right now—tap "+" to add any place!');
    } finally {
      setIsAutoDetectingLocation(false);
    }
  };

      });

      source.connect(analyser);

      audioContextRef.current = audioCtx;
      analyserRef.current = analyser;

      const dataArray = new Uint8Array(analyser.frequencyBinCount);
      const updateMeter = () => {
        analyser.getByteFrequencyData(dataArray);
        let sum = 0;
        for (let i = 0; i < dataArray.length; i++) sum += dataArray[i];
        setAudioLevel(0.95 + Math.min(0.45, (sum / dataArray.length) / 130));
        animFrameRef.current = requestAnimationFrame(updateMeter);
      };
      updateMeter();
    } catch (e) {}

            }
          ]

      `;

      if (!SpeechRec) return;

      if (recognitionRef.current) {
        try { recognitionRef.current.onend = null; recognitionRef.current.abort(); } catch {}

      const recognition = new SpeechRec();
      recognition.continuous = true;
      recognition.interimResults = true;

      const clerkLangObj = LANGUAGE_LIST.find(l => l.code === clerkLangCode);
      recognition.lang = clerkLangObj ? clerkLangObj.recognitionCode : 'zh-TW';

        let currentInterim = '';
        let highestConfidence = 0;

        for (let i = event.resultIndex; i < event.results.length; i++) {
          const result = event.results[i][0];
          currentInterim += result.transcript;
          if (event.results[i].isFinal && result.confidence > highestConfidence) {
            highestConfidence = result.confidence;
          }

        }

                <Text style={styles.attentionBtnText}>{t.attentionBtn}</Text>

                style={[styles.gpsScanBtn, { borderColor: activeVenue.accent }]}

                  <Text style={[styles.floatingBubbleText, bubble.isCore && { color: activeVenue.accent }]}>

            {predictiveSuggestions.length > 0 && (

                  </TouchableOpacity>
                </View>
              )}

              <TouchableOpacity style={styles.cancelBtn} onPress={() => { setIsEditingSOS(false); setIsSOSModalOpen(false); }}>
                <Text style={[styles.cancelBtnText, { color: '#FFF' }]}>{t.cancel}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

        {/* MODAL 3: Edit Venue Studio (Change Name, Emoji, Color, Icon & Background Image URL) */}
        <Modal visible={isEditVenueModalOpen} transparent animationType="slide">
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>🎨 Customize "{activeVenue.short}"</Text>

                <Text style={styles.sectionHeader}>Custom Icon Image URL (Optional)</Text>

                  style={styles.modalTextInput}
                  value={editVenueBgUrl}

                  placeholderTextColor="rgba(255,255,255,0.35)"

              <TouchableOpacity style={[styles.saveBtn, { backgroundColor: editVenueAccent }]} onPress={handleSaveEditedVenue}>

              </TouchableOpacity>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setIsEditVenueModalOpen(false)}>
                <Text style={styles.cancelBtnText}>{t.cancel}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

        {/* MODAL 4: Language Settings */}
        <Modal visible={isSettingsOpen} transparent animationType="slide">
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>{t.modalTitle}</Text>
              <ScrollView style={{ maxHeight: 380 }}>
                <Text style={styles.sectionHeader}>{t.staffLangLabel}</Text>
                {LANGUAGE_LIST.map((lang) => (
                  <TouchableOpacity
                    key={`clerk-${lang.code}`}
                    style={[styles.languageOption, clerkLangCode === lang.code && styles.languageOptionSelected]}
                    onPress={() => setClerkLangCode(lang.code)}
                  >
                    <Text style={styles.languageOptionText}>{lang.name}</Text>
                    {clerkLangCode === lang.code && <Text style={{ color: '#30D158', fontWeight: 'bold' }}>✓</Text>}
                  </TouchableOpacity>
                ))}
                <Text style={[styles.sectionHeader, { marginTop: 18 }]}>{t.userLangLabel}</Text>
                {LANGUAGE_LIST.map((lang) => (
                  <TouchableOpacity
                    key={`user-${lang.code}`}
                    style={[styles.languageOption, userLangCode === lang.code && styles.languageOptionSelected]}
                    onPress={() => setUserLangCode(lang.code)}
                  >
                    <Text style={styles.languageOptionText}>{lang.name}</Text>
                    {userLangCode === lang.code && <Text style={{ color: '#FF8A1F', fontWeight: 'bold' }}>✓</Text>}
                  </TouchableOpacity>
                ))}
              </ScrollView>
              <TouchableOpacity style={styles.closeModalButton} onPress={() => setIsSettingsOpen(false)}>
                <Text style={styles.closeModalText}>{t.saveSettings}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

        {/* MODAL 5: Add New Place with AI */}
        <Modal visible={isAddLocationModalOpen} transparent animationType="fade">
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>{t.addLocationModalTitle}</Text>
              <Text style={styles.modalSub}>{t.addLocationModalSub}</Text>
              
              <TextInput
                style={styles.modalTextInput}
                placeholder={t.addLocationPlaceholder}
                placeholderTextColor="rgba(255,255,255,0.4)"
                value={newLocationInput}
                onChangeText={setNewLocationInput}
                onSubmitEditing={handleGenerateNewVenue}
                autoFocus
              />

              <TouchableOpacity style={[styles.saveBtn, isGeneratingVenue && { opacity: 0.5 }]} disabled={isGeneratingVenue} onPress={handleGenerateNewVenue}>
                {isGeneratingVenue ? (
                  <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8 }}>
                    <ActivityIndicator color="#FFF" size="small" />
                    <Text style={styles.saveBtnText}>{t.addLocationAnalyzingBtn}</Text>
                  </View>
                ) : (
                  <Text style={styles.saveBtnText}>{t.addLocationAnalyzeBtn}</Text>
                )}
              </TouchableOpacity>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setIsAddLocationModalOpen(false)}>
                <Text style={styles.cancelBtnText}>{t.cancel}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

      </View>
    </View>
  );
};
