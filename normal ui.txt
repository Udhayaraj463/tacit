        if (highestConfidence > 0 && highestConfidence < 0.5) return; 

        setLiveStreamText(text);

        if (debounceTimerRef.current) clearTimeout(debounceTimerRef.current);
        
        debounceTimerRef.current = setTimeout(() => {
          dispatchLiveIntent(text);
        }, 1200);
      };

      recognition.onerror = () => {};
      recognition.onend = () => { if (isLiveActive) try { recognition.start(); } catch {} };

      recognitionRef.current = recognition;
      if (isLiveActive) try { recognition.start(); } catch {}
    }
    return () => { if (debounceTimerRef.current) clearTimeout(debounceTimerRef.current); };
  }, [clerkLangCode, isLiveActive, activeVenue.id, isOnline]);

  const toggleLiveSession = () => {
    if (!isLiveActive) {
      setIsLiveActive(true);
      startAudioMeter();
      try { recognitionRef.current?.start(); } catch {}
    } else {
      setIsLiveActive(false);
      stopAudioMeter();
      interruptAudio();
      try { recognitionRef.current?.stop(); } catch {}
      setLiveStreamText('');
    }
  };

  const handleSelectOption = (opt: ActionOption) => {
    interruptAudio();
    setClerkDisplayMessage(opt.labelClerk);
    speakText(opt.labelClerk, clerkLangCode);
  };

  const handleSelectBubble = (bubble: ChitChatBubble) => {
    interruptAudio();
    const clerkPhrase = bubble.clerkText[clerkLangCode] || bubble.clerkText['zh-TW'];
    setClerkDisplayMessage(clerkPhrase);
    speakText(clerkPhrase, clerkLangCode);

    if (bubble.followUps && bubble.followUps.length > 0) {
      setActiveBubbles([...CORE_REACTIONS, ...bubble.followUps]);
    }
  };

  const handleGenerateNewVenue = async () => {
    const input = newLocationInput.trim();
    if (!input || isGeneratingVenue) return;

    if (!isOnline) {
      alert('Cannot analyze locations while offline.');
      return;
    }

    setIsGeneratingVenue(true);
    try {
      const generated = await fetchDynamicVenueCards(input);

      if (generated && generated.options) {
        const updatedVenues = [generated, ...venues];
        setVenues(updatedVenues);
        setActiveVenue(generated);

        const customOnly = updatedVenues.filter(v => !DEFAULT_VENUES.some(d => d.id === v.id));
        await AsyncStorage.setItem('@aac_custom_venues', JSON.stringify(customOnly));

        setNewLocationInput('');
        setIsAddLocationModalOpen(false);
      }
    } catch (err) {
      alert('Unable to analyze location.');
    } finally {
      setIsGeneratingVenue(false);
    }
  };

  const resolveAssetSource = (src: any) => {
    if (!src) return undefined;
    if (typeof src === 'string') return { uri: src };
    return src;
  };

  const clerkLang = LANGUAGE_LIST.find(l => l.code === clerkLangCode) || LANGUAGE_LIST[0];
  const userLang = LANGUAGE_LIST.find(l => l.code === userLangCode) || LANGUAGE_LIST[0];

  return (
    <View style={styles.outerScreen}>
      
      <View style={[StyleSheet.absoluteFill, { zIndex: 0, backgroundColor: isScreenFlashing ? '#FFF' : '#05070C' }]}>
        <Image 
          source={resolveAssetSource(activeVenue.bgImage)}
          style={{ width: '100%', height: '100%', resizeMode: 'cover', opacity: isScreenFlashing ? 0 : 0.65 }}
          blurRadius={3}
        />
        <div style={{ position: 'absolute', inset: 0, background: 'linear-gradient(to bottom, rgba(5,7,12,0.7) 0%, rgba(5,7,12,0.6) 50%, rgba(5,7,12,0.95) 100%)', opacity: isScreenFlashing ? 0 : 1 }} />
      </View>

      <View style={[styles.mobileContainer, { paddingTop: insets.top, paddingBottom: insets.bottom }]}>

        <View style={styles.clerkRegion}>
          <div style={glassOverlayStyle} />
          <View style={styles.clerkRotatedContainer}>
            <Text style={styles.clerkMetaTag}>[ {t.clerkViewTag} • {activeVenue.short.toUpperCase()} ]</Text>
            <Text style={[styles.clerkResponseText, { color: activeVenue.accent }]} numberOfLines={3} adjustsFontSizeToFit>
              {clerkDisplayMessage}
            </Text>

            <TouchableOpacity style={styles.clerkKeyboardToggle} onPress={() => setIsClerkKeyboardOpen(true)}>
              <Text style={styles.clerkKeyboardToggleText}>{t.clerkTypeBtn}</Text>
            </TouchableOpacity>
          </View>
        </View>

        <View style={styles.divider}>
          <div style={glassOverlayStyle} />
          <View style={styles.liveIndicatorRow}>
            <View style={[styles.liveDot, isLiveActive && styles.liveDotActive]} />
            <Text style={styles.dividerContextLabel}>
              {isLiveActive ? t.live : t.standby} • {clerkLang.short} ➔ {userLang.short}
            </Text>
          </View>
          <TouchableOpacity style={styles.settingsPill} onPress={() => setIsSettingsOpen(true)}>
            <Text style={styles.settingsPillText}>{t.languages}</Text>
          </TouchableOpacity>
        </View>

        <View style={styles.contentRow}>
          <View style={styles.userRegion}>

            {!isOnline && (
              <View style={styles.offlineNoticeBar}>
                <Text style={styles.offlineNoticeText}>{t.offlineNotice}</Text>
              </View>
            )}

            <View style={{ flexDirection: 'row', gap: 6, marginBottom: 6 }}>
              <TouchableOpacity style={styles.attentionBtn} onPress={triggerAttention}>
                <Text style={styles.attentionBtnText}>{t.attentionBtn}</Text>
              </TouchableOpacity>
            </View>

            <View style={styles.kineticContainer}>
              <div style={glassOverlayStyle} />
              <BallStudy mode="dark" scale={audioLevel} brightness={isLiveActive ? (isSpeaking ? 1.6 : 1.35) : 0.75} />
              {liveStreamText ? (
                <View style={styles.liveTranscriptOverlay}>
                  <Text style={styles.liveStreamText} numberOfLines={2}>"{liveStreamText}"</Text>
                </View>
              ) : null}
            </View>

            <TouchableOpacity style={[styles.liveButton, isLiveActive ? styles.liveButtonOn : styles.liveButtonOff]} onPress={toggleLiveSession}>
              <Text style={styles.liveButtonText}>{isLiveActive ? t.endLive : t.startLive}</Text>
            </TouchableOpacity>

            <View style={styles.bubblesHeaderRow}>
              <Text style={styles.bubblesHeaderTitle}>{t.floatingBubblesTitle}</Text>
              <TouchableOpacity onPress={() => setActiveBubbles(CHIT_CHAT_SYSTEM)}>
                <Text style={styles.bubblesResetText}>{t.reset}</Text>
              </TouchableOpacity>
            </View>
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.bubbleScrollRow}>
              {activeBubbles.map((bubble) => (
                <TouchableOpacity
                  key={bubble.id}
                  style={[styles.floatingBubble, bubble.isCore && { backgroundColor: 'rgba(255,255,255,0.15)', borderWidth: 1, borderColor: activeVenue.accent }]}
                  onPress={() => handleSelectBubble(bubble)}
                >
                  <div style={glassOverlayStyle} />
                  <Text style={[styles.floatingBubbleText, bubble.isCore && { color: activeVenue.accent }]}>
                    {bubble.labels[userLangCode] || bubble.labels['en-US']}
                  </Text>
                </TouchableOpacity>
              ))}
            </ScrollView>

            {predictiveSuggestions.length > 0 && (
              <ScrollView horizontal showsHorizontalScrollIndicator={false} style={{ marginBottom: 4 }}>
                {predictiveSuggestions.map((sug, i) => (
                  <TouchableOpacity key={i} style={styles.predictivePill} onPress={() => { setCustomTypedInput(sug); }}>
                    <Text style={styles.predictivePillText}>{sug}</Text>
                  </TouchableOpacity>
                ))}
              </ScrollView>
            )}

            <View style={styles.customInputRow}>
              <div style={glassOverlayStyle} />
              <TextInput
                style={styles.customInputField}
                placeholder={isTranslating ? t.translatingText : t.typePlaceholder}

                value={customTypedInput}
                onChangeText={handleTyping}
                onSubmitEditing={handleCustomTypedSubmit}

              </TouchableOpacity>
            </View>

            {liveHeardTranslated ? (
              <View style={styles.transcriptBox}>
                <div style={glassOverlayStyle} />
                <View style={styles.transcriptHeader}>
                  <Text style={styles.transcriptLabel}>{t.understoodIntent}</Text>
                </View>
                <Text style={styles.transcriptText} numberOfLines={2}>{liveHeardTranslated}</Text>
              </View>
            ) : null}

            <View style={styles.sectionHeaderRow}>
              <Text style={styles.quickReplyTitle}>{t.liveLocationText}</Text>
              <View style={[styles.locationTagPill, { borderColor: activeVenue.accent, flexDirection: 'row', alignItems: 'center', gap: 4 }]}>
                {isAutoDetectingLocation && <ActivityIndicator size="small" color={activeVenue.accent} />}
                <Text style={[styles.locationTagText, { color: activeVenue.accent }]}>📍 {activeVenue.nameKey}</Text>
              </View>
            </View>

            <ScrollView contentContainerStyle={styles.actionCardGrid} showsVerticalScrollIndicator={false}>
              {currentOptions.map((opt) => (
                <TouchableOpacity key={opt.id} style={[styles.actionCard, { borderLeftColor: activeVenue.accent }]} onPress={() => handleSelectOption(opt)}>
                  <div style={glassOverlayStyle} />
                  <Text style={[styles.actionCardText, { color: activeVenue.accent }]}>{opt.labelUser}</Text>
                  <Text style={styles.previewSubtext}>{t.facingClerkPrefix} "{opt.labelClerk}" 🔊</Text>
                </TouchableOpacity>
              ))}
            </ScrollView>
          </View>

          <View style={styles.sidebarDock}>
            <div style={dockGlassStyle} />
            <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.dockScroll}>
              {venues.map((loc) => {
                const isSelected = activeVenue.id === loc.id;
                return (
                  <TouchableOpacity
                    key={loc.id}
                    style={[styles.dockImageBox, isSelected && [styles.dockImageBoxActive, { borderColor: loc.accent }]]}
                    onPress={() => { setActiveVenue(loc); if (loc.id === 'hospital') setSelectedHospitalDept('all'); }}
                  >
                    <Image source={resolveAssetSource(loc.iconImage)} style={{ width: '100%', height: '100%', resizeMode: 'cover' }} />
                  </TouchableOpacity>
                );
              })}

                <Text style={styles.addLocationText}>＋</Text>
              </TouchableOpacity>
              
              <TouchableOpacity style={styles.sosSidebarButton} onPress={() => setIsSOSModalOpen(true)}>
                <Text style={styles.sosSidebarText}>{t.sosBtn}</Text>
              </TouchableOpacity>
            </ScrollView>
          </View>

        </View>

          <View style={[styles.modalOverlay, { justifyContent: 'center' }]}>
            <View style={[styles.modalContent, { transform: [{ rotate: '180deg' }] }]}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>{t.clerkTypeBtn}</Text>
              <TextInput

                placeholder={t.clerkTypePlaceholder}

                value={clerkTypedInput}

                autoFocus
              />
              <TouchableOpacity style={styles.saveBtn} onPress={() => {
                setLiveHeardTranslated(`(Typed by Clerk): ${clerkTypedInput}`);
                setIsClerkKeyboardOpen(false);
                setClerkTypedInput('');
              }}>
                <Text style={styles.saveBtnText}>{t.clerkSend}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setIsClerkKeyboardOpen(false)}>

              </TouchableOpacity>
            </View>
          </View>

        <Modal visible={isSOSModalOpen} transparent animationType="slide">
          <View style={[styles.modalOverlay, { backgroundColor: 'rgba(255, 59, 48, 0.95)' }]}>
            <View style={styles.sosModalContent}>
              <Text style={styles.sosModalTitle}>🚨 {t.sosTitle} 🚨</Text>
              <View style={styles.sosInfoBox}>
                <Text style={styles.sosInfoText}>Name: <Text style={{fontWeight:'900'}}>Udhayaraj (Raj)</Text></Text>
                <Text style={styles.sosInfoText}>Blood: <Text style={{fontWeight:'900'}}>O+</Text></Text>
                <Text style={styles.sosInfoText}>Allergies: <Text style={{fontWeight:'900'}}>None</Text></Text>
                <Text style={styles.sosInfoText}>Emergency: <Text style={{fontWeight:'900'}}>+886 123 456 789</Text></Text>
              </View>
              <TouchableOpacity 
                style={styles.sosSpeakActionBtn} 
                onPress={() => speakText("I am having a medical emergency and cannot speak. Please read my screen.", clerkLangCode)}
              >
                <Text style={styles.sosSpeakActionText}>{t.sosSpeak}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setIsSOSModalOpen(false)}>
                <Text style={styles.cancelBtnText}>{t.cancel}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

        <Modal visible={isSettingsOpen} transparent animationType="slide">
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>{t.modalTitle}</Text>
              <ScrollView style={{ maxHeight: 380 }}>
                <Text style={styles.sectionHeader}>{t.staffLangLabel}</Text>
                {LANGUAGE_LIST.map((lang) => (
                  <TouchableOpacity
                    key={`clerk-${lang.code}`}
                    style={[styles.languageOption, clerkLangCode === lang.code && styles.languageOptionSelected]}
                    onPress={() => setClerkLangCode(lang.code)}
                  >
                    <Text style={styles.languageOptionText}>{lang.name}</Text>
                    {clerkLangCode === lang.code && <Text style={{ color: '#30D158', fontWeight: 'bold' }}>✓</Text>}
                  </TouchableOpacity>
                ))}
                <Text style={[styles.sectionHeader, { marginTop: 18 }]}>{t.userLangLabel}</Text>
                {LANGUAGE_LIST.map((lang) => (
                  <TouchableOpacity
                    key={`user-${lang.code}`}
                    style={[styles.languageOption, userLangCode === lang.code && styles.languageOptionSelected]}
                    onPress={() => setUserLangCode(lang.code)}
                  >
                    <Text style={styles.languageOptionText}>{lang.name}</Text>
                    {userLangCode === lang.code && <Text style={{ color: '#FF8A1F', fontWeight: 'bold' }}>✓</Text>}
                  </TouchableOpacity>
                ))}
              </ScrollView>
              <TouchableOpacity style={styles.closeModalButton} onPress={() => setIsSettingsOpen(false)}>
                <Text style={styles.closeModalText}>{t.saveSettings}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

        <Modal visible={isAddLocationModalOpen} transparent animationType="fade">
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <div style={glassOverlayStyle} />
              <Text style={styles.modalTitle}>{t.addLocationModalTitle}</Text>
              <Text style={styles.modalSub}>{t.addLocationModalSub}</Text>
              
              <TextInput
                style={styles.modalTextInput}
                placeholder={t.addLocationPlaceholder}
                placeholderTextColor="rgba(255,255,255,0.4)"
                value={newLocationInput}
                onChangeText={setNewLocationInput}
                autoFocus
              />

              <TouchableOpacity style={[styles.saveBtn, isGeneratingVenue && { opacity: 0.5 }]} disabled={isGeneratingVenue} onPress={handleGenerateNewVenue}>
                {isGeneratingVenue ? (
                  <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8 }}>
                    <ActivityIndicator color="#FFF" size="small" />
                    <Text style={styles.saveBtnText}>{t.addLocationAnalyzingBtn}</Text>
                  </View>
                ) : (
                  <Text style={styles.saveBtnText}>{t.addLocationAnalyzeBtn}</Text>
                )}
              </TouchableOpacity>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setIsAddLocationModalOpen(false)}>
                <Text style={styles.cancelBtnText}>{t.cancel}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </Modal>

      </View>
    </View>
  );
};
