// Universal Muted Video / GIF Renderer (Works with .mp4, .webm, .mov, .gif, data:video, blob:, or require())
const MochiMediaPlayer: React.FC<{ source: any; style?: any }> = ({ source, style }) => {
  let uri = '';
  if (typeof source === 'string') {
    uri = source;
  } else if (source && typeof source === 'object' && source.uri) {
    uri = source.uri;
  } else if (source) {
    const resolved = Image.resolveAssetSource ? Image.resolveAssetSource(source) : null;
    uri = resolved?.uri || String(source);
  }

  const isVideo =
    typeof uri === 'string' &&
    (/\.(mp4|webm|mov|ogg)(\?.*)?$/i.test(uri) || uri.startsWith('data:video/') || uri.includes('video'));

  if (isVideo && typeof document !== 'undefined') {
    return (
      <video
        key={uri}
        src={uri}
        autoPlay
        loop
        muted
        playsInline
        style={{
          width: '100%',
          height: '100%',
          objectFit: 'cover',
          pointerEvents: 'none',
          backgroundColor: '#FFF',
          ...(style || {})
        }}
      />
    );
  }

  return (
    <Image
      source={typeof source === 'string' ? { uri: source } : source}
      style={[{ width: '100%', height: '100%', resizeMode: 'cover' }, style]}
    />
  );
};

// Helper to convert Gemini's 24kHz PCM Base64 audio into a playable browser WAV Blob
const createWavUrlFromPcmBase64 = (base64Pcm: string, sampleRate = 24000): string => {
  const binaryStr = atob(base64Pcm);
  const len = binaryStr.length;
  const buffer = new ArrayBuffer(44 + len);
  const view = new DataView(buffer);

  const writeStr = (offset: number, str: string) => {
    for (let i = 0; i < str.length; i++) view.setUint8(offset + i, str.charCodeAt(i));
  };

  writeStr(0, 'RIFF');
  view.setUint32(4, 36 + len, true);
  writeStr(8, 'WAVE');
  writeStr(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true); // PCM
  view.setUint16(22, 1, true); // Mono
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeStr(36, 'data');
  view.setUint32(40, len, true);

  for (let i = 0; i < len; i++) {
    view.setUint8(44 + i, binaryStr.charCodeAt(i));
  }

  const blob = new Blob([buffer], { type: 'audio/wav' });
  return URL.createObjectURL(blob);
};

  const speakText = async (text: string, langTarget: SupportedLang) => {
    if (typeof window === 'undefined' || !text.trim()) return;
    interruptAudio();

    setIsSpeaking(true);
    setMochiMood('speaking');
    setMochiGifPhase('speaking');

    // @ts-ignore

        const ttsRes = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [{ parts: [{ text }] }],
              generationConfig: {
                responseModalities: ['AUDIO'],
                speechConfig: {
                  voiceConfig: {
                    prebuiltVoiceConfig: {
                      voiceName: 'Ursa'
                    }
                  }
                }
              }
            })
          }
        );

        if (ttsRes.ok) {
          const ttsData = await ttsRes.json();
          const inlineAudio = ttsData?.candidates?.[0]?.content?.parts?.[0]?.inlineData;
          if (inlineAudio?.data) {
            const wavUrl = createWavUrlFromPcmBase64(inlineAudio.data, 24000);
            const audio = new Audio(wavUrl);
            activeAudioObjRef.current = audio;
            audio.onended = () => {
              setIsSpeaking(false);
              setMochiMood(isLiveActiveRef.current ? 'listening' : 'happy');
              setMochiGifPhase(isMochiHubOpen ? 'venue' : 'idle');
            };
            audio.onerror = () => {
              setIsSpeaking(false);
              setMochiGifPhase(isMochiHubOpen ? 'venue' : 'idle');
            };
            await audio.play();
            return; // Successfully spoke with Gemini Ursa voice!
          }
        }
      } catch (err) {
        console.warn('Gemini Ursa TTS fallback to device voice:', err);
      }
    }

    // 2. Instant Offline / Fallback Sweet Voice
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      const found = LANGUAGE_LIST.find(l => l.code === langTarget);
      const targetTag = found ? found.ttsLang : langTarget;
      u.lang = targetTag;

      const voices = window.speechSynthesis.getVoices();
      const langPrefix = targetTag.split('-')[0].toLowerCase();
      const matchingVoices = voices.filter(v => v.lang.toLowerCase().includes(langPrefix));
      const sweetVoice = matchingVoices.find(v => 
        /natural|aria|jenny|samantha|mei-jia|hsiao|kyoko|google|female|siri|zira/i.test(v.name)
      ) || matchingVoices[0];

      if (sweetVoice) u.voice = sweetVoice;
      u.pitch = 1.15;
      u.rate = 1.02;

      u.onend = () => {
        setIsSpeaking(false);
        setMochiMood(isLiveActiveRef.current ? 'listening' : 'happy');
        setMochiGifPhase(isMochiHubOpen ? 'venue' : 'idle');
      };
      u.onerror = () => {
        setIsSpeaking(false);
        setMochiMood('happy');
        setMochiGifPhase(isMochiHubOpen ? 'venue' : 'idle');
      };

          const result = event.results[i][0];
          currentInterim += result.transcript;
          if (event.results[i].isFinal && result.confidence > highestConfidence) {
            highestConfidence = result.confidence;
          }
        }

        const text = currentInterim.trim();
        if (!text) return;
        if (highestConfidence > 0 && highestConfidence < 0.45) return; 

        setLiveStreamText(text);
        setMochiMood('listening');

        if (debounceTimerRef.current) clearTimeout(debounceTimerRef.current);
        debounceTimerRef.current = setTimeout(() => {
          dispatchLiveIntent(text);
        }, 1000);
      };

      recognition.onerror = () => {};
      recognition.onend = () => {
        if (isLiveActiveRef.current) {
          try { recognition.start(); } catch {}
        }
      };

      recognitionRef.current = recognition;
      if (isLiveActive) {
        try { recognition.start(); } catch {}
      }
    }
    return () => {
      if (debounceTimerRef.current) clearTimeout(debounceTimerRef.current);
    };
